# FONT (font.c): despite its name, the introduction's story scenes ("The Summoning" ...: a four-row strip of map tiles with
# actors, played from a script in MISCMAPS.DAT), the justified proportional text printer, and the character creation
# (name, sex, the gypsy's questions that set intelligence, dexterity and strength). One overlay.
# functions
FONT_0000 FONT_0000_DrawJustifiedText    # asm in the DOS build (AFONT0.ASM): prints justified text with the proportional font
FONT_02a2 FONT_02a2_DrawStripCell        # draws the strip's cell (x, y): the map tile or else the actor's tile
FONT_02fc FONT_02fc_AnimateScene         # runs the scene's animation for so many frames (strip widening, sounds); 1 when a key was pressed
FONT_0418 FONT_0418_LoadSceneStrip       # a scene's title and its four rows of map from the scratch buffer
FONT_04a4 FONT_04a4_PlayIntroStory       # "view": the intro's story scenes, run from the script at 0x200 of MISCMAPS.DAT (actors placed, moved, moongates, lightning, loops)
FONT_0998 FONT_0998_PickVirtue           # a random virtue (0..7) not offered before and not rejected; marks it offered
FONT_09c8 FONT_09c8_AskVirtueQuestion    # the gypsy's question for two virtues: A or B decides the stat bonuses; the other virtue is out
FONT_0b0a FONT_0b0a_CreateCharacter      # asks the name and sex, the seven questions, sets the stats and writes the new saved game
FONT_0e52 FONT_0e52_SaveVideoPage        # asm (AFONT.ASM): saves the video page buffer before the tile set changes
FONT_0e7b FONT_0e7b_RestoreVideoPage     # asm (AFONT.ASM): and restores it
