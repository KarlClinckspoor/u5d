# COMSUBS (comsubs.c): the combat subroutines shared by the melee, missile, spell and monster code:
# who is where, the line of a missile, its pictures, hit reports. One overlay.
# functions
COMSUBS_0000 COMSUBS_0000_TargetResists       # intelligence contest of two combatants; 1 when the target holds out
COMSUBS_0056 COMSUBS_0056_FadeFields          # each field tile (E8-EB) on the map has 16 in 256 to vanish
COMSUBS_00d2 COMSUBS_00d2_PrintMiss           # "Failed!" for a spell, else "<name> missed!"
COMSUBS_00f4 COMSUBS_00f4_MonsterSpecial      # a monster's own tricks: possess, vanish and return, gate in a daemon; 1 when used
COMSUBS_0312 COMSUBS_0312_ReportHit           # prints what a blow did (grazed, killed, slept, wounded..) by the flags in D_58a2
COMSUBS_0458 COMSUBS_0458_DistanceSquared
COMSUBS_048a COMSUBS_048a_Distance            # integer square root of the above, by subtracting 1, 3, 5..
COMSUBS_04d4 COMSUBS_04d4_CombatantDistance
COMSUBS_0504 COMSUBS_0504_AimCrosshair        # the crosshair moves by the arrow keys within a range; returns the distance, 0 when cancelled
COMSUBS_0748 COMSUBS_0748_CombatantAt         # the standing combatant on a cell, else -1
COMSUBS_07d4 COMSUBS_07d4_PickNearbyCell      # a random cell within one step of (x, y), on the 11x11 map, into D_5876/D_5878
COMSUBS_0822 COMSUBS_0822_ResolveShot         # flies a missile at a cell (scattered when it missed); returns the combatant it hits, else -1
COMSUBS_097c COMSUBS_097c_SpendAmmo           # a shot uses an arrow or quarrel; a thrown weapon is used up
COMSUBS_09fc COMSUBS_09fc_EnemyInterferes     # an adjacent enemy that last attacked this one spoils the spell or shot
COMSUBS_0a68 COMSUBS_0a68_FireMissile         # a ranged attack: aim, fly, hit or miss
COMSUBS_0bf8 COMSUBS_0bf8_MeleeBlow           # a hand-to-hand blow: roll to hit, damage, report
COMSUBS_0c52 COMSUBS_0c52_AttackWith          # the attack with one weapon: melee when its range is 0, else a missile
COMSUBS_0d3c COMSUBS_0d3c_AttackWithWeapon    # one readied weapon's attack (named first when several)
COMSUBS_0d96 COMSUBS_0d96_AttackCmd           # the Attack command: bare hands or each readied weapon
COMSUBS_0e26 COMSUBS_0e26_TracePath           # the pixels of a straight line, 100 units a pixel, ending in 0xff
COMSUBS_0f4a COMSUBS_0f4a_DrawMissile         # one frame of a missile (dart, boulder, bolt, field sparkle..) by its kind
COMSUBS_12de COMSUBS_12de_FlyMissile          # animates a missile along the path; 0 when something stops it
