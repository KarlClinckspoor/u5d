# TOWN (town.c, town2.c): the rest: the town map, party movement, alarm, shadowlords, the main loop.
# (TOWN_0000, 0052, 00b0 and the named commands were named in the pilot.)
# functions
TOWN_011e TOWN_011e_FindNpcByActor          # the NPC slot whose actor is the given one; -1 none
TOWN_0170 TOWN_0170_SwapDayNightTiles       # lamps (tile 0x87) get the tile below toggled; the annotated tiles (0x48/0x49) show at 5 o'clock, else water
TOWN_0212 TOWN_0212_BlightLand              # with a shadowlord in town most trees and crops are cut (a fixed random sequence by the day)
TOWN_02ae TOWN_02ae_SetupShadowlord         # which shadowlord haunts this town, and his actor and NPC slot
TOWN_0408 TOWN_0408_LoadTownMap             # loads the map of this level, lists the annotations and beacons, night tiles, the light map
TOWN_052e TOWN_052e_TakeStairs              # the stairs just stepped on go up or down, by the facing
TOWN_057c TOWN_057c_SetPartyIcon            # the party's tile by the way it faces, with the "Ride/Fly/Row" word
TOWN_0600 TOWN_0600_MoveParty               # a step in a direction (the town's version); returns 1 when the party leaves the map
TOWN_085e TOWN_085e_SetPursue               # schedule: no times, type 6 or 7 (pursue the Avatar)
TOWN_08d4 TOWN_08d4_SetFlee                 # a villager (type 0x40-0x73): schedule type 3 (flee), dialog 0xfd
TOWN_0958 TOWN_0958_RaiseAlarm              # guards, shadowlords and the like pursue; other NPCs flee half the time
TOWN_09bc TOWN_09bc_FightNpc                # the NPC is marked killed, the party fights its actor, the town is rebuilt
TOWN_0c4a TOWN_0c4a_IsTileA2Or43            # horses do not wander next to these tiles
TOWN_0c78 TOWN_0c78_WanderHorses            # the loose horses of the town take a random step
TOWN_0dc4 TOWN_0dc4_GetTownKey              # update the screen, read a key; a drunk party staggers (Hic!)
TOWN_0e34 TOWN_0e34_NumberKey               # 1-6 at the instrument tile 0x8d plays a note (a tune opens a way); else selects the player
TOWN_0f02 TOWN_0f02_TileEffects             # the tile stood on: sleepers wake, trapdoor, swamp poison, fire; ends the turn
TOWN_10da TOWN_10da_ShoutBegone             # "Begone, vermin!": the party is damaged, the NPC flees
TOWN_10f2 TOWN_10f2_PickNpcForAlarm         # a villager with a schedule time, half the time
TOWN_1156 TOWN_1156_ShadowlordEffect        # hatred: NPCs pursue (dialog 0xfe); cowardice: they flee (0xfd)
TOWN_11b8 TOWN_11b8_PrintAirOf              # "An air of falsehood/hatred/cowardice doth surround thee..."
TOWN_12ae TOWN_12ae_Arrest                  # "Thou art under arrest!": comes quietly (jail) or fights; 1 when fighting
TOWN_1352 TOWN_1352_NpcApproaches           # an NPC next to the Avatar attacks, talks, or arrests him
TOWN_1694 TOWN_1694_PlaceActors             # clears the NPC actors and puts each NPC on its place by the hour
TOWN_1726 TOWN_1726_PlaceNpcActor           # the NPC's actor on this level, or none on another; its position
