ZSTATS_099a ZSTATS_099a_FillItemCounts            # copies the saved game's items into the D_b9ee table (the Use list)
ZSTATS_0bee ZSTATS_0bee_PrintEquipRefusal         # a refusal message, then the "Item:" prompt again
ZSTATS_0c0a ZSTATS_0c0a_FreeHandSlot              # 2 both hands free, 0 first hand free, 1 second, 0xff none
ZSTATS_0c5c_EquipItem ZSTATS_0c5c_EquipOrRemove   # ready or take off an item; 1 when a ring vanished
ZSTATS_0f2e ZSTATS_0f2e_SelectItem                # the cursor list of Ready (mode 0x52) and Use; returns the item or -1
