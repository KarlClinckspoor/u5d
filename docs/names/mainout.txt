# MAINOUT (mainout.c + mainoutb.c): the overworld: the main command loop, walking and sailing, entering towns,
# dungeons and shrines, and the creatures that roam it (spawning, moving, attacking). One overlay: mainout.c is MSC 5.1,
# mainoutb.c (from MAINOUT_109e) MSC 5.0.
# functions
MAINOUT_0000 MAINOUT_0000_SetUpOverworld       # enters the overworld: clears the sky, sets the view origin around the party, loads the map chunks, the light map, wind, sun and moons, the underworld's items
MAINOUT_007a MAINOUT_007a_WaitIfActorNear      # one frame's wait when an actor of the table is within five cells (only actor 1 is looked at: the loop breaks at once)
MAINOUT_00da MAINOUT_00da_TurnVehicle          # "Ride/Fly/Row/Head": turns the party's tile to the direction; 1 when the party only turned (no step), also a frigate under no wind
MAINOUT_01fe MAINOUT_01fe_CanStepTo            # whether the party may step by (x, y): the actor there and the walkable tile decide; prints Blocked!, OUCH!, COLLISION!, BREAKING UP!, Docked!
MAINOUT_0354 MAINOUT_0354_MoveParty            # the party moves by (x, y); near the edge of the 32x32 view the chunks are shifted and loaded
MAINOUT_03e0 MAINOUT_03e0_SlowTerrain          # after a step: on rough terrain (forest, hills, mountains) the monsters move once or twice more, "Slow progress!" / "Very slow!"
MAINOUT_0490 MAINOUT_0490_WalkInDirection      # a direction key: prints the direction, turns the vehicle, steps when allowed (and the frigate sets sail); 1 when it stepped
MAINOUT_0598 MAINOUT_0598_WaitForKey           # redraws, handles the falling tile, prints the prompt; a frigate under sail moves with the wind until a key; returns the key
MAINOUT_0a1a MAINOUT_0a1a_UpdateDarkness       # on a blank (0xff) tile without the status 0xe the light level is 0, the view is redrawn once
MAINOUT_0a60 MAINOUT_0a60_CheckEarthquake      # underground: one in 256 turns "EARTHQUAKE!" damages the party
MAINOUT_0d8c MAINOUT_0d8c_SpawnChance          # how likely a monster spawn is, 0 (towns, castles) to 5: by the tile under the party, + 3 at night; 3 in the underworld
MAINOUT_0e04 MAINOUT_0e04_PickWeighted         # an index picked by weights that add up to 256 (a roll of 0-255 minus each weight)
MAINOUT_0e4e MAINOUT_0e4e_MonsterForTile       # the monster (actor tile) that may appear on that terrain: sea, land or underworld tables, whirlpools in deep water; 0 for none
MAINOUT_0f4e MAINOUT_0f4e_PickSpawnCell        # a random cell of the 32x32 view more than six away from the party, into D_5876/D_5878
MAINOUT_0fc4 MAINOUT_0fc4_SpawnMonster         # looks up to 128 times for a cell and a monster that fit, and puts it there (pirates get hull value 100)
MAINOUT_105c MAINOUT_105c_IsMovingActor        # whether an actor tile is one that moves by itself: 0x2c-0x2f and 0x80 up except 0xb4-0xb7 and 0xe8-0xeb
MAINOUT_109e MAINOUT_109e_DamageVehicle        # a blow to the party's vehicle: a ship loses 1 to 30 hull points and sinks (abandon ship, or drowning); on foot the party is hurt
MAINOUT_1168 MAINOUT_1168_PirateFire           # a pirate ship fires (cannon sound, a missile to the party); a hit hurts the vehicle
MAINOUT_1248 MAINOUT_1248_ActorTouchesParty    # an actor next to the party: a whirlpool pulls the ship into the underworld, else "Attacked!" and combat (not on a skiff or carpet over water)
MAINOUT_131a MAINOUT_131a_ActorAttack          # next to the party: touches it; sea monsters fire from three cells at 1 in 8; a pirate ship in line says "BOOOM!"; 1 when it acted
MAINOUT_1482 MAINOUT_1482_CanActorEnter        # the cell is walkable for the actor and holds no other actor
MAINOUT_14c8 MAINOUT_14c8_NotLastCell          # 0 for the cell the last moved actor came from (D_a526/D_a527)
MAINOUT_14ea MAINOUT_14ea_InChaseRange         # the actor is within six cells and in the pattern D_2c18
MAINOUT_1578 MAINOUT_1578_MoveActor            # moves an actor by (x, y): a pirate ship turns to face it; rough terrain may stop it; it leaves a moongate's cell empty
MAINOUT_16fc MAINOUT_16fc_MoveRandomly         # tries random directions, up to three times, and takes the first free cell
MAINOUT_17d4 MAINOUT_17d4_MoveTowardParty      # one step toward the party, along x or y by chance; otherwise at random (the second parameter is only written)
MAINOUT_198c MAINOUT_198c_ActorTurn            # one actor's move by kind: a whirlpool every other turn, the Shadowlord after the party (20 moves), a ship by the wind, else toward the party
MAINOUT_1a60 MAINOUT_1a60_MonstersTurn         # the monsters' turn after a step: may spawn one, each moving actor attacks or moves, those out of the view vanish; the number that attacked
MAINOUT_1b3e MAINOUT_1b3e_TrollToll            # the trolls ask for a toll (99 - 3 x strength of the first, gold); not paid, they fight
MAINOUT_1be8 MAINOUT_1be8_BridgeTrolls         # crossing a bridge: one in eight, each party member's dexterity tests whether the trolls under it see them
#= MAINOUT_06ec_AttackCmd # the Attack command: a direction, then combat with what is there, if it can be fought (not on a skiff or carpet over water)
#= MAINOUT_0790_LoadDungeon # prints what is entered, finds the dungeon at the party's position (D_1e8a/D_1eb2, entries 0x20-0x27), saves the actors and reads its DUNGEON.DAT level; Doom's entrance may be guarded

#= MAINOUT_0d22_Entry # the overlay's entry: sets up the overworld, puts a bought ship at its place, runs the main loop
