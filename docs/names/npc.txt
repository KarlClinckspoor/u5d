# NPC (npc.c + ANPC.ASM): the NPCs of a town: loading, schedules, walking, path search.
# functions
NPC_00d6 NPC_00d6_PlaceNpcs                 # on entering a town: every NPC to the schedule entry in force at the hour, mode 1
NPC_01a0 NPC_01a0_FindLadder                # path search for the nearest ladder (up, or down); its tile ends in D_5876/D_5878
NPC_01d2 NPC_01d2_BuildSearchGrid           # fills D_b11c (32x32): 0 free, 0x90 blocked, 5 goal, 0x46 start
NPC_032c NPC_032c_SearchPath                # breadth-first search over D_b11c, direction in the high nibble; 1 when the goal is reached
NPC_04ac NPC_04ac_BuildMoveList             # walks the marks back from the goal and stores (steps, direction) runs, reversed
NPC_0632 NPC_0632_StepTemp                  # moves (D_5876, D_5878) one tile: 1 east, 2 north, 3 west, 4 south
NPC_06a0 NPC_06a0_Distance                  # manhattan distance
NPC_06e4 NPC_06e4_PursueOrFlee              # one step towards the Avatar (away for type 3); next to him it asks to talk (0x74) or attacks (0x61)
NPC_0938 NPC_0938_ScheduleDue               # a schedule time has come: sets the NPC's mode (walk, change level); nonzero when it must move
NPC_0a4a NPC_0a4a_StandsOnStairs            # does the NPC stand on a ladder or stairs towards the level of its schedule entry
NPC_0adc NPC_0adc_CanEnter                  # the tile is the goal (2), or free by the table D_367e (1), else 0
NPC_0b9e NPC_0b9e_CanStepTo                 # the tile is on the map, the NPC may enter it and no actor stands there
NPC_0c50 NPC_0c50_Wander                    # one random step, kept within maxDist of the schedule's place
NPC_0d00 NPC_0d00_ActBySchedule             # acts by the schedule entry's type: stand, wander, follow, pursue
NPC_0db4 NPC_0db4_MoveNpcs                  # the per-turn movement of every NPC of the town
NPC_12e0 NPC_12e0_ScheduleSlot              # which of the three schedule entries is in force at the hour (assembly, ANPC.ASM)
